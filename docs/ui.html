private fun loadHealthConnect(showLoading: Boolean, forceDataRead: Boolean = false) {
    viewModelScope.launch {
        if (showLoading) _uiState.update { it.copy(isLoading = true) }

        val previousStatus = _uiState.value.healthConnect
        val status = getHealthConnectPermissionStatusUseCase()
        _uiState.update { it.copy(healthConnect = status, error = null) }

        // ⭐ Chỉ đọc lại khi: lần nạp đầu, được yêu cầu rõ ràng, HOẶC trạng thái quyền ĐÃ ĐỔI
        val shouldReadData = showLoading || forceDataRead || status != previousStatus

        if (status.status != HealthConnectStatus.AVAILABLE) {
            // Không có Health Connect: xoá số liệu cũ; nguồn "dữ liệu trong app" vẫn đọc được
            _uiState.update { it.copy(healthData = HealthData(), isLoading = false) }
            if (shouldReadData) loadStoredHealthData()
            return@launch
        }

        if (shouldReadData) readActiveHealthData()
    }
}

// Nguồn do switch "Dùng dữ liệu Health Connect" quyết định:

private suspend fun readActiveHealthData() {

2.3. Hộp thoại "Khung giờ ngủ"

Một trong số ít hộp thoại có bản nháp trong state — cần thiết vì người dùng chọn giờ, sửa lại, rồi mới lưu:

fun openSleepScheduleEditor() = _uiState.update { it.copy(sleepScheduleDraft = it.sleepSchedule, sleepScheduleError = null) }
fun dismissSleepScheduleEditor() = _uiState.update { it.copy(sleepScheduleDraft = null, sleepScheduleError = null ) }
fun onSleepScheduleDraftChange(draft: SleepSchedule) = _uiState.update { it.copy(sleepScheduleDraft = draft, sleepScheduleError = null) }

fun saveSleepSchedule() {
    val draft = _uiState.value.sleepScheduleDraft ?: return

    // Khung không hợp lệ (giờ đi ngủ TRÙNG giờ thức) → KHÔNG ghi gì, giữ hộp thoại mở + báo lỗi
    if (!draft.isValid) {
        _uiState.update { it.copy(sleepScheduleError = SLEEP_SCHEDULE_ERROR_EMPTY) }
        return
    }
    viewModelScope.launch {
        settingsRepository.setSleepSchedule(draft)
        _uiState.update { it.copy(sleepScheduleDraft = null, sleepScheduleError = null) }
    }
}

3.1. Trạng thái & tiến độ

data class HabitWithProgress(val habit: Habit, val progress: Float, val isCompleted: Boolean)

Nguồn là combine của hai Flow: danh sách thói quen đang hoạt động và nhật ký của hôm nay (getHabitLogsForPeriod(0)). Tiến độ = tổng value của các log cùng habitId trong ngày; isCompleted khi tổng ≥ targetValue.

Vì sao daysBack = 0 nghĩa là "hôm nay"

HabitRepositoryImpl.getHabitLogsForPeriod(daysBack) tính mốc là 00:00 của daysBack ngày trước, không phải now − daysBack. Nếu tính là now − 0 thì mốc bằng đúng lúc gọi hàm ⇒ mọi log đã ghi trong ngày đều bị loại và tiến độ/trạng thái hoàn thành về 0 mỗi lần mở lại màn hình. Đây là lỗi cũ, nay đã sửa và ghi chú rõ trong KDoc.

3.2. Bộ lọc theo buổi

Bộ lọc là một enum, không phải chuỗi tiếng Việt viết cứng:

enum class HabitFilter(@StringRes val labelRes: Int, val timeOfDay: HabitTimeOfDay?) {
    ALL(R.string.habits_filter_all, null),
    MORNING(R.string.habits_filter_morning, HabitTimeOfDay.MORNING),
    AFTERNOON(R.string.habits_filter_afternoon, HabitTimeOfDay.AFTERNOON),
    EVENING(R.string.habits_filter_evening, HabitTimeOfDay.EVENING)
}

// Lọc theo buổi + LUÔN giữ thói quen "Cả ngày" (không gắn với buổi cụ thể)
private fun applyFilter(habits: List<HabitWithProgress>, filter: HabitFilter): List<HabitWithProgress> {
    val timeOfDay = filter.timeOfDay ?: return habits
    return habits.filter { it.habit.timeOfDay == timeOfDay || it.habit.timeOfDay == HabitTimeOfDay.ALL_DAY }
}

3.3. Ghi nhận giá trị — chặn ngay ở UI và ở ViewModel

// presentation/screen/habits/HabitLogInput.kt — hàm THUẦN, test được trên JVM
fun parseLogValue(input: String): LogValueResult {
    val trimmed = input.trim()
    if (trimmed.isEmpty()) return LogValueResult.Invalid("Vui lòng nhập số lượng")

    val value = trimmed.toFloatOrNull() ?: return LogValueResult.Invalid("Số lượng phải là số")
    if (!value.isFinite()) return LogValueResult.Invalid("Số lượng phải là số")   // NaN / Infinity
    if (value <= 0f) return LogValueResult.Invalid("Số lượng phải lớn hơn 0")

    return LogValueResult.Valid(value)
}

ViewModel vẫn kiểm lại — lưới an toàn cho mọi đường vào khác:

    }
}

            lastJpeg = jpeg                    // giữ để nút "Thử lại" chạy lại không cần chụp lại
            when (val result = foodInsightSource.analyze(jpeg)) {
                is FoodAnalysisResult.Success -> _uiState.update {
                    it.copy(
                        isAnalyzing = false,
                        insight = result.insight.takeIf { i -> i.hasDish },
                        // "không có đồ ăn" HOẶC "có nhưng không gọi được tên món" → coi như chưa nhận diện
                        noFoodDetected = !result.insight.hasDish,
                        error = null, canRetry = false
                    )
                }
                is FoodAnalysisResult.Failure -> _uiState.update {
                    it.copy(isAnalyzing = false, error = messageFor(result.error), canRetry = lastJpeg != null)
                }
            }
        } catch (e: CancellationException) {
            throw e                         // rời màn không phải lỗi phân tích
        } catch (e: Exception) {
            _uiState.update { it.copy(error = "Lỗi phân tích món ăn: ${e.message}", isAnalyzing = false, canRetry = lastJpeg != null) }
        }
    }
}

Một hộp thoại không tắt được bằng Back hay chạm ra ngoài (vòng xoay + mô tả), để người dùng biết AI đang làm việc — cần thiết vì một lượt phân tích có thể mất vài giây.

enum class HeartRateGuidance { READY_TO_START, PLACE_FINGER, WEAK_SIGNAL, MEASURING, ALMOST_DONE }

// Câu hướng dẫn được SUY RA từ trạng thái đo, không phải một chuỗi cố định
val guidance: HeartRateGuidance
    get() = when {
        phase != HeartRatePhase.MEASURING -> HeartRateGuidance.READY_TO_START
        !fingerDetected -> HeartRateGuidance.PLACE_FINGER                        // "Chưa thấy ngón tay"
        signalStrength < MIN_SIGNAL_STRENGTH -> HeartRateGuidance.WEAK_SIGNAL     // "Tín hiệu yếu"
        elapsedMs < HeartRateMath.MIN_MEASURE_MS -> HeartRateGuidance.MEASURING    // "Đang đo nhịp tim"
        else -> HeartRateGuidance.ALMOST_DONE                                     // "Sắp xong"
    }

/** Tiến độ vòng tròn: đã đo được bao phần của thời gian TỐI THIỂU (0..1) */
val progress: Float
    get() = (elapsedMs.toFloat() / HeartRateMath.MIN_MEASURE_MS).coerceIn(0f, 1f)

UI gồm: thẻ hướng dẫn 3 bước trước khi đo; trong lúc đo là thẻ hướng dẫn đổi theo tình trạng + vòng tròn tiến độ 15–30 giây + số nhịp hiện dần + dạng sóng mạch đập + thanh "Tín hiệu" (Mạnh / Trung bình / Yếu).

7.2. Bốn nguồn dữ liệu hợp nhất

private fun observeHealthData() {
    combine(
        getHealthInsightsUseCase(),          // gợi ý AI (Rule Engine)
        getVitalityScoreUseCase(),           // chỉ số sinh lực
        getWeeklyConsistencyUseCase(),       // biểu đồ tuần
        getRecentHealthSummaryUseCase()      // lịch sử 7 ngày
    ) { suggestions, vitality, consistency, healthHistory ->
        InsightsSnapshot(suggestions, vitality, consistency, healthHistory)
    }
        .catch { e ->
            // ⚠ Huỷ scope KHÔNG phải lỗi dữ liệu → ném tiếp thay vì hiện thẻ lỗi
            if (e is CancellationException) throw e
            android.util.Log.e(TAG, "Không đọc được dữ liệu phân tích", e)
            _uiState.update { it.copy(isLoading = false, error = "Không đọc được dữ liệu phân tích: ${e.message ?: "lỗi không xác định"}") }
        }
        .onEach { snapshot ->
            _uiState.update {
                it.copy(
                    suggestions = snapshot.suggestions,
                    vitality = snapshot.vitality,
                    weeklyConsistency = snapshot.weeklyConsistency,
                    sleepTrend = SleepTrendMath.fromHealthHistory(snapshot.healthHistory),   // dẫn xuất
                    healthHistory = snapshot.healthHistory,
                    isLoading = false,
                    error = null                        // đọc được dữ liệu ⇒ xoá thẻ lỗi cũ
                )
            }
        }.launchIn(viewModelScope)
}

sleepTrend được tính từ chính healthHistory thay vì đọc riêng — nhờ vậy thẻ giấc ngủ dùng cùng nguồn số với bảng bên dưới và với Rule Engine.

private suspend fun readSummary(entryPoint: HealthWidgetEntryPoint): TodayHealthSummary {
    /* … */
    val sensorSteps = if (stepCounterStatus.isReady) {
        // NO_WAIT_MS: widget chỉ cần giá trị ĐÃ LƯU, không chờ mẫu cảm biến mới
        // (chờ một mẫu có thể mất tới 2 giây và làm chậm cả lần vẽ widget)
        stepCounter.readTodaySteps(StepCounterRepository.NO_WAIT_MS)
    } else 0L

    return TodayHealthSummary.build(
        healthConnect = todayHealthData().getOrNull(),
        sensorStepsReady = stepCounterStatus.isReady,
        sensorSteps = sensorSteps,

        recordedHeartRate = latestRecordedHeartRate()

}

  • 230 <string> trong app/src/main/res/values/strings.xml.
